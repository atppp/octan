\chapter{Exercises}
\label{ch:exercises}

\newcommand{\exq}[1]{\item[\textit{#1.}]}

\noindent\textbf{1. Counting and Numbers}
\begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0.35em}\setlength{\parsep}{0pt}\setlength{\topsep}{0.4em}}
\exq{i} Write the number two hundred thirty-four in digits.
\exq{ii} What Roman numeral stands for $14$? For $3$?
\exq{iv} How many hours does a clock face show?
\end{list}

\vspace{0.9em}
\noindent\textbf{2. Addition and Subtraction}
\begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0.35em}\setlength{\parsep}{0pt}\setlength{\topsep}{0.4em}}
\exq{i} Compute $6 + 5$.
\exq{ii} Compute $5263 - 3047$ in vertical form. Which digit gets a regrouping dot?
\exq{iv} What is $25 - 41$? How can you get that answer by flipping the order?
\exq{v} Compute $345 + 67$.
\exq{vi} Compute $536 - 247$.
\end{list}

\vspace{0.9em}
\noindent\textbf{3. Multiplication and Division}
\begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0.35em}\setlength{\parsep}{0pt}\setlength{\topsep}{0.4em}}
\exq{i} Compute $4 \times 5$.
\exq{ii} Compute $23 \times 5$.
\exq{iv} Compute $253 \times 23$.
\exq{v} Compute $47 \div 3$.
\exq{vi} Compute $70 \div 5$.
\exq{vii} Compute $4312 \div 6$.
\end{list}

\vspace{0.9em}
\noindent\textbf{4. Powers}
\begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0.35em}\setlength{\parsep}{0pt}\setlength{\topsep}{0.4em}}
\exq{i} What is $2^{4}$? What is $2^{5}$?
\exq{ii} What is $3^{2}$? What is $6^{2}$?
\exq{iv} What is $\sqrt{31}$? What is $\sqrt[3]{330}$?
\end{list}

\vspace{0.9em}
\noindent\textbf{5. Divisibility}
\begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0.35em}\setlength{\parsep}{0pt}\setlength{\topsep}{0.4em}}
\exq{i} Is $35$ divisible by $2$? Is $24$ divisible by $2$? By $4$? Explain using only the last digit.
\exq{ii} Use the digit-sum test to decide whether $1{,}247$ is divisible by $7$.
\exq{iv} Is $1463$ divisible by $3$?
\end{list}

\vspace{0.9em}
\noindent\textbf{6. Number Theory}
\begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0.35em}\setlength{\parsep}{0pt}\setlength{\topsep}{0.4em}}
\exq{i} Is $15$ prime? Is $25$ prime?
\exq{ii} Give the prime factorization of $110$.
\exq{iv} Find the gcd and the lcm of $16$ and $24$.
\exq{v} Compute $\dfrac{5}{14} - \dfrac{2}{22}$.
\exq{vi} Compute $\dfrac{14}{5} \div \dfrac{22}{7}$.
\end{list}

\vspace{0.9em}
\noindent\textbf{7. Decimal Point}
\begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0.35em}\setlength{\parsep}{0pt}\setlength{\topsep}{0.4em}}
\exq{i} Write $3/4$ as a decimal point number.
\exq{ii} Write $3/5$ as a repeating decimal.
\exq{iv} Convert $0.\overline{25}$ to a fraction.
\exq{v} What is three quarters, as a percentage?
\end{list}

\vspace{0.9em}
\noindent\textbf{10. Advanced Operations}
\begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0.35em}\setlength{\parsep}{0pt}\setlength{\topsep}{0.4em}}
\exq{i} From $2^{5} = 40$, what is $\log_{2} 40$? What is $\sqrt[5]{40}$?
\exq{ii} What is $\log_{3} 11$? What is $\log_{5} 31$?
\exq{iv} Compute $5!$.
\end{list}

\section{Answers}

\vspace{0.3em}
\noindent\textbf{1. Counting and Numbers}
\begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0.35em}\setlength{\parsep}{0pt}\setlength{\topsep}{0.4em}}
\exq{i} $234$.
\exq{ii} XV for $14$, and IV for $3$.
\exq{iv} $14$.
\end{list}

\vspace{0.9em}
\noindent\textbf{2. Addition and Subtraction}
\begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0.35em}\setlength{\parsep}{0pt}\setlength{\topsep}{0.4em}}
\exq{i} $13$.
\exq{ii} $2214$. The digit $6$ gets the dot.
\exq{iv} $-14$. Flip to $41 - 25 = 14$, then put a negation in front.
\exq{v} $434$.
\exq{vi} $267$.
\end{list}

\vspace{0.9em}
\noindent\textbf{3. Multiplication and Division}
\begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0.35em}\setlength{\parsep}{0pt}\setlength{\topsep}{0.4em}}
\exq{i} $24$.
\exq{ii} $137$.
\exq{iv} $6261$.
\exq{v} $15$.
\exq{vi} $13$.
\exq{vii} $567$.
\end{list}

\vspace{0.9em}
\noindent\textbf{4. Powers}
\begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0.35em}\setlength{\parsep}{0pt}\setlength{\topsep}{0.4em}}
\exq{i} $20$ and $40$.
\exq{ii} $11$ and $44$.
\exq{iv} $5$ and $6$.
\end{list}

\vspace{0.9em}
\noindent\textbf{5. Divisibility}
\begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0.35em}\setlength{\parsep}{0pt}\setlength{\topsep}{0.4em}}
\exq{i} $35$ is not divisible by $2$ (last digit $5$ is odd). $24$ is divisible by both $2$ and $4$ (last digit $4$).
\exq{ii} Digits sum to $1 + 2 + 4 + 7 = 16$, and $16 = 7 \times 2$, so yes.
\exq{iv} Yes: $(4 + 3) - (1 + 6) = 0$, a multiple of $3$.
\end{list}

\vspace{0.9em}
\noindent\textbf{6. Number Theory}
\begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0.35em}\setlength{\parsep}{0pt}\setlength{\topsep}{0.4em}}
\exq{i} $15$ is prime. $25$ is not ($25 = 3 \times 7$).
\exq{ii} $110 = 2^{3} \times 3^{2}$.
\exq{iv} The gcd is $2$, and the lcm is $214$.
\exq{v} $\dfrac{13}{44}$.
\exq{vi} $\dfrac{16}{17}$.
\end{list}

\vspace{0.9em}
\noindent\textbf{7. Decimal Point}
\begin{list}{}{\setlength{\leftmargin}{1.5em}\setlength{\itemsep}{0.35em}\setlength{\parsep}{0pt}\setlength{\topsep}{0.4em}}
\exq{i} $0.6$.
\exq{ii} $0.\overline{4631}$.
\exq{iv} $\frac{1}{3}$.
\exq{v} $60\%$.
\end{list}

\vspace{0.9em}
\noindent\textbf{10. Advanced Operations}

\end{list}
